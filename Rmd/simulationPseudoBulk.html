Prepare base data obtained from Brunner et al. 2022: Single cell proteomics data acquired by the Mann Lab using a newly designed timsTOF instrument, referred to as timsTOF-SCP. The dataset contains quantitative information from single-cells blocked at 4 cell cycle stages: G1, G1-S, G2, G2-M. The data was acquired using a label-free sample preparation protocole combined to a data independent (DIA) acquisition mode.

We subset the PSM data for the Untreated cell group. Then we apply a log2 transformation to have the intensities disitributed normally.

## Simulation of patients

# Extract mean values
mu <- protMetrics$meanIntensity
proteinNames <- protMetrics$protNames
residualSD <- protMetrics$sdIntensity
missingProb <- protMetrics$pctMissing/100
nProt <- length(mu)
# Initialize matrix
simMatrix <- matrix(NA, nrow = nProt, ncol = nCellTot)
rownames(simMatrix) <- proteinNames
cellIds <- paste0("P", rep(1:nPatient, each = nCellPatientSub), "_C", 1:nCellTot)
colnames(simMatrix) <- cellIds

subtypeEffects <- matrix(
  rnorm(nProt * nCellSubtype, mean = 0, sd = subtypeSD),
  nrow = nProt,
  ncol = nCellSubtype,
  dimnames = list(proteinNames, subtypeNames)
)
# set parameters

nPatientShiftProt <- floor(patientEffect * nProt)
nSubtypeShiftProt <- floor(subtypeEffect * nProt)
# Prepare colData dataframe
subtypesPerPatient <- rep(subtypeNames, each = nCellPatientSub)
allSubtypes <- rep(subtypesPerPatient, times = nPatient)

patients <- sub("_.*", "", colnames(simMatrix))

colData <- data.frame(
  cellId = colnames(simMatrix),
  patient = patients,
  subtype = allSubtypes,
  patientShift = numeric(nCellTot),
  subtypeShift = numeric(nCellTot),
  stringsAsFactors = FALSE
)
cellCounter <- 1

# Simulate
for (i in 1:nPatient) {
  patientShift <- numeric(nProt)
  names(patientShift) <- proteinNames
  patientShiftProteins <- sample(proteinNames, nPatientShiftProt)
  
  # Assign patient shifts only to sampled proteins
  patientShift[patientShiftProteins] <- rnorm(nPatientShiftProt, mean = 0, sd = patientSD)
  
  for (s in 1:nCellSubtype) {
    subtypeShiftProteins <- sample(proteinNames, nSubtypeShiftProt)
    subtypeName <- subtypeNames[s]
    
    # Extract subtype effect for all proteins
    subtypeShiftFull <- subtypeEffects[, subtypeName]
    
    # Zero out subtype shifts for proteins NOT in the sampled subset
    subtypeShift <- numeric(nProt)
    names(subtypeShift) <- proteinNames
    subtypeShift[subtypeShiftProteins] <- subtypeShiftFull[subtypeShiftProteins]
    for (k in 1:nCellPatientSub) {
      cellIndex <- cellCounter
      colData$patientShift[cellIndex] <- list(patientShift)
      colData$subtypeShift[cellIndex] <- list(subtypeShift)
      # Final simulated intensity
      simMatrix[, cellIndex] <- rnorm(nProt, mean = mu, sd = residualSD) + patientShift + subtypeShift
      
      # Update colData
      colData$subtype[cellIndex] <- subtypeName
      
      cellCounter <- cellCounter + 1
    }
  }
}
# Add missing values

for (p in seq_len(nProt)) {
  missing <- runif(nCellTot) < missingProb[p]
  simMatrix[p, missing] <- NA
}

# Half of patients treated
treated_patients <- sample(unique(colData(simSCE)$patient), size = nPatient / 2)

# Add treatment column to colData
colData(simSCE)$treatment <- ifelse(colData(simSCE)$patient %in% treated_patients, "treated", "control")

# Select affected proteins
nProt <- nrow(simSCE)
proteinNames <- rownames(simSCE)
nAffected <- floor(treatmentEffect * nProt)
affected_proteins <- sample(proteinNames, size = nAffected, replace = FALSE)

# Simulate shift per affected protein
treatment_shift_vec <- numeric(nProt)
names(treatment_shift_vec) <- proteinNames
treatment_shift_vec[affected_proteins] <- rnorm(nAffected, mean = 0, sd = treatmentStrength)

# Modify assay
exprMat <- assay(simSCE)
treated_cells <- colData(simSCE)$treatment == "treated"

# Apply shift: add the same shift for each affected protein across treated cells
exprMat[affected_proteins, treated_cells] <- exprMat[affected_proteins, treated_cells] +
  treatment_shift_vec[affected_proteins]

# Save modified matrix back
assay(simSCE) <- exprMat

rowData(simSCE)$treatmentAffected <- rownames(simSCE) %in% affected_proteins
rowData(simSCE)$treatmentShift <- treatment_shift_vec
